# 아이템 생성 / 타입

> [!NOTE]
>
> 이 페이지는 현재 작성 중이며 최근 `JSON INFO`에서 분리되었습니다.

### 일반 아이템

```json
"type": "GENERIC",                // 일반 아이템으로 정의
"id": "socks",                    // 고유 ID. 공백 없이 하나의 단어로, 필요시 밑줄 사용
"name": {
    "ctxt": "clothing",           // 선택적 번역 컨텍스트. 문자열에 여러 의미가 있어 다른 언어로 번역할 때 다르게 번역해야 할 때 유용
    "str": "pair of socks",       // 조사 상자에 표시되는 이름. 공백으로 구분된 여러 단어 가능
    "str_pl": "pairs of socks"    // 선택사항. 이름이 불규칙한 복수형을 가진 경우(즉, 단수형에 "s"를 붙여 복수형을 만들 수 없는 경우) 지정해야 함. 단수와 복수가 같은 경우 "str_sp" 사용 가능
},
"conditional_names": [ {          // 지정된 조건에서 적용될 이름의 선택적 목록 (자세한 내용은 조걶� 명명 섹션 참조).
    "type": "COMPONENT_ID",       // 조건 타입.
    "condition": "leather",       // 확인할 조건.
    "name": { "str": "pair of leather socks", "str_pl": "pairs of leather socks" } // 이름 필드, 위와 같은 규칙 적용
} ],
"container": "null",             // 이 아이템이 생성되어야 하는 컨테이너(있는 경우)
"repairs_like": "scarf",          // 이 아이템에 제작법이 없는 경우, 수리할 때 어떤 아이템의 제작법을 찾을지
"color": "blue",                 // 아이템 심볼의 색상
"symbol": "[",                   // 맵에 표시되는 아이템 심볼. 정확히 1개의 콘솔 셀 너비를 가진 유니코드 문자열이어야 함
"looks_like": "rag",              // 타일셋에 힌트. 이 아이템에 타일이 없는 경우 looks_like 타일 사용
"description": "Socks. Put 'em on your feet.", // 아이템 설명
"ascii_picture": "ascii_socks", // 이 아이템에 사용되는 asci_art의 ID
"phase": "solid",                            // (선택사항, 기본값 = "solid") 상태
"spill_field": "fd_water",                   // (선택사항) 액체를 일반 지면에 떨어뜨릴 때 drop_action 전에 생성할 필드입니다.
"weight": "350 g",                           // 무게, 그램 단위로 표현, mg과 kg 사용 가능 - "50 mg", "5 g" 또는 "5 kg". 쌓을 수 있는 아이템(탄약, 식량)의 경우 이것은 충전량당 무게
"volume": "250 ml",                          // 부피, ml과 L 단위 사용 가능 - "50 ml" 또는 "2 L". 쌓을 수 있는 아이템(탄약, 식량)의 경우 이것은 stack_size 충전량의 부피
"integral_volume": 0,                        // 아이템이 다른 아이템에 통합될 때(예: 총에 통합된 총기 개조) 기본 아이템에 추가되는 부피. ml과 L 단위 사용 가능 - "50 ml" 또는 "2 L". 부모 부피를 줄이기 위해 음수일 수 있음. 부모 기본 부피의 1%로 제한됨
"integral_weight": 0,                        // 아이템이 다른 아이템에 통합될 때(예: 총에 통합된 총기 개조) 기본 아이템에 추가되는 무게. 부모 무게를 줄이기 위해 음수일 수 있음. 부모 기본 무게의 1%로 제한됨
"rigid": false,                              // 비강성 아이템의 경우 부피(및 착용 아이템의 경우 동작 제한)는 내용물에 비례하여 증가
"insulation": 1,                             // (선택사항, 기본값 = 1) 컨테이너나 차량 부품인 경우 내용물에 제공해야 하는 보온 정도
"price": 100,                                // NPC와 거래할 때 사용. 쌓을 수 있는 아이템(탄약, 식량)의 경우 이것은 stack_size 충전량당 가격. 문자열 "cent" "USD" 또는 "kUSD" 사용 가능
"price_postapoc": "1 USD",                   // price와 같지만 대재앙 이후 가치를 나타냄. 문자열 "cent" "USD" 또는 "kUSD" 사용 가능
"material": ["COTTON"],                      // 재료 타입, 원하는 만큼 많이 사용 가능. 가능한 옵션은 materials.json 참조
"weapon_category": [ "WEAPON_CAT1" ],        // (선택사항) 이 아이템이 속한 무술 무기 카테고리
"cutting": 0,                                // (선택사항, 기본값 = 0) 근접 무기로 사용할 때 가하는 절단 피해. 음수 값은 불가능
"bashing": 0,                                // (선택사항, 기본값 = 0) 근접 무기로 사용할 때 가하는 타격 피해. 음수 값은 불가능
"to_hit": 0,                                 // (선택사항, 기본값 = 0) 근접 무기로 사용할 때 명중 별림치
"attacks": [                                 // (선택사항) 새로운 공격 스탯블록, 진행 중인 기능
  { "id": "BASH",                            // 공격의 ID. ID가 "DEFAULT"인 공격은 계산된 데이터로 대첼됨("copy-from" 아이템에서 커스텀 공격을 제거하는 데 사용 가능)
    "to_hit": 1,                             // 이 공격의 명중 별림치
    "damage": { "values": [ { "damage_type": "bash", "amount": 50 } ] } }, // 이 공격의 피해, `damage_instance` 구문 사용 (아래 참조)
  { "id": "THRUST", "damage": { "values": [ { "damage_type": "stab", "amount": 45 } ] } }
],
"flags": ["VARSIZE"],                        // 특수 효과를 나타냄, json_flags.md 참조
"environmental_protection_with_filter": 6,   // 아이템(예: 방독면)에 필터가 필요하고 필터가 설치된 경우 환경 효과에 대한 저항. use_action 'GASMASK' 및 'DIVE_TANK'와 함께 사용
"magazine_well": 0,                          // 탄창이 아이템에서 튀어나와 추가 부피를 더하기 시작하는 부피
"magazines": [                               // 각 탄약 타입(있는 경우)에 사용할 수 있는 탄창 타입
    [ "9mm", [ "glockmag" ] ]                // 각 탄약 타입에 대해 지정된 첫 번째 탄창이 기본값
    [ "45", [ "m1911mag", "m1911bigmag" ] ],
],
"crafting_speed_modifier": 1.0,              // 선택 사항(기본값 = 1.0). 이 아이템이 도구 또는 품질 요구사항을 충족할 때 제작 속도에 적용할 배율
"milling": {                                 // 선택사항. 주어진 경우 아이템을 물/풍차에서 분쇄할 수 있음
  "into": "flour",                           // 제품의 아이템 ID. 제품은 반드시 충전량을 사용하는 것이어야 함
  "conversion_rate": 4                       // 소비된 아이템당 제품 수. conversion_rate가 4인 경우 1개 아이템을 4개 제품으로 분쇄. 정수만 허용
},
"item_vars": {                               // 아이템 변수 미리 채우기; 현재는 모드와 게임 내 로직 외에는 사용되지 않음
  "test": "test"                             // 주로 lua 모드에서 유용함
}
"explode_in_fire": true,                     // 불에 태울 때 폭발해야 하는지
"explosion": {                               // 물리적 폭발 데이터
  "damage": 10,                              // 폭발 중심지에서 플레이어에게 가하는 피해. 50% 반경 이상에서는 피해가 절반으로 감소
  "radius": 8,                               // 폭발 반경. 0은 중심지만 영향받음을 의미
  "fire": true,                              // 폭발이 불을 남겨야 하는지
  "fragment_effect": [ {                     // "파편"에 적용할 효과 데이터
      "effect": "onfire",                   // 적용할 효과(onfire에는 대상의 가연성을 확인하는 특수 동작이 있음)
      "odds": 2,                              // 이 효과가 적용될 1/x 확률
      "min_turns": 4,                         // 효과의 최소 지속 턴
      "max_turns": 8                          // 효과의 최대 지속 턴
    }
  ],
  "fragment": {                              // "파편"의 투사체 데이터. 이 투사체는 범위와 시야 내의 모든 대상에 정확히 한 번 명중
    "damage": {                              // 파편 투사체의 피해 데이터. `damage_instance` 구문 사용 (아래 참조)
      "damage_type": "acid",                 // 가하는 피해 타입
      "amount": 10                           // 가하는 피해량
      "armor_penetration": 4                 // 무시하는 방어력. 방어구별로 적용되며 총계가 아님
      "armor_multiplier": 2.5                // 방어력에서 남은 피해 감소에 적용되는 승수, 관통 후 적용(있는 경우). 높은 숫자는 방어구가 이 폭발의 파편을 더 효과적으로 막게 하고, 낮은 숫자는 남은 방어력의 백분율 감소로 작동
    }
  }
},
"repair_difficulty": 2                       // 수리 난이도를 덮어쓰는 제작법 난이도
```

`spill_field`는 액체를 위한 필드입니다. `spill_field`가 있는 액체 아이템을 일반 지면에 떨어뜨리면 필요할 때 액체를 더럽힌 뒤 지정된 필드를 만들고, 아이템이 정의한 `drop_action`도 계속 실행합니다. 액체 아이템은 소비되며 충전량과 아이템 상태는 필드에 저장되지 않습니다. 웅덩이를 다시 액체로 회수할 수도 없습니다. `LIQUIDCONT` 플래그가 있는 타일에서는 액체를 스필 필드로 변환하지 않습니다.

스필의 시각 효과는 충전량이 아니라 부피를 기준으로 합니다. 현재 구현에서는 액체 1리터를 보이는 스필 강도 1로 취급하며, 0이 아닌 스필에는 최소 1의 강도를 적용합니다. 같은 필드 유형의 연결된 스필에 액체를 추가하면 먼저 기존 타일의 필드 강도를 높여 깊게 만든 뒤 인접한 통행 가능 타일로 확장합니다. 보이는 면적에는 상한이 있으므로 매우 큰 스필은 시각적으로 더 커지지 않더라도 부어진 액체는 계속 소비합니다.

실제로 `spill_field`는 보통 여러 액체를 하나의 필드에 대응시키는 데 사용합니다. 관련된 여러 액체를 같은 보이는 필드에 의도적으로 대응하면 JSON 중복과 불필요한 필드 유형 증가를 피할 수 있습니다. 예를 들어 `tea`, `coca_tea`, `sweet_tea`는 모두 `fd_tea`로 스필되어 같은 `puddle of tea` 필드를 만들 수 있습니다. 웅덩이의 효과나 외관이 다를 때만 별도의 필드 유형이 필요합니다. 필드 속성은 [필드 유형](../map/field_type.md)을 참조하세요.

#### damage_instance

```json
{
  "damage_type": "acid", // 가하는 피해 타입
  "amount": 10, // 가하는 피해량
  "armor_penetration": 4, // 무시하는 방어력. 방어구별로 적용되며 총계가 아님
  "armor_multiplier": 2.5 // 방어력에서 남은 피해 감소에 적용되는 승수, 관통 후 적용(있는 경우). 높은 숫자는 방어구가 더 효과적으로 보호하게 하고, 낮은 숫자는 남은 방어력의 백분율 감소로 작동
}
```

### 탄약

```json
"type" : "AMMO",            // 탄약으로 정의
...                         // 일반 아이템과 동일한 항목
                            // 추가적인 탄약 특정 항목:
"ammo_type" : "shot",       // 장전할 수 있는 것을 결정
"damage" : 18,              // 발사 시 원거리 피해
"prop_damage": 2,           // 무기 피해에 승수 적용(damage 필드 덮어씀)
"pierce" : 0,               // 발사 시 방어 관통 능력
"range" : 5,                // 발사 시 사거리
"dispersion" : 0,           // 탄약의 부정확성, 1/4도 단위로 측정
"recoil" : 18,              // 발사 시 발생하는 반동
"count" : 25,               // 함께 생성되는 탄 수
"stack_size" : 50,          // (선택사항) 위에서 정의한 부피에 몇 발이 들어있는지. 생략하면 'count'와 동일
"show_stats" : true,        // (선택사항) 전투 탄약에 대해 강제로 스탯 표시(damage와 prop_damage가 모두 없는 투사체용)
"dont_recover_one_in": 1    // (선택사항) 탄약을 회수하지 않을 1/x 확률(100은 99% 회수 확률 의미)
"drop": "nail",             // (선택사항) 투사체 위치에 100% 확률로 떨어지는 객체 정의
"drop_active": false        // (선택사항) 객체가 활성화된 상태로 시작하는지. 기본값은 true
"drop_count": 1,            // (선택사항) 떨어뜨릴 아이템 수. 도구의 경우 이것은 충전량을 설정
                             // 생략하면 해당 타입의 'count'에서 정의한 떨어뜨림 양이 기본값
"shot": {                   // (선택사항) 펠릿형 탄약의 산탄 패턴 데이터
  "count": 12,             // 한 발에서 생성되는 투사체 공격 수
  "half_angle": 3          // 펠릿 퍼짐과 미리보기 원뿔에 쓰이는 반각(도 단위)
},
"effects" : ["COOKOFF", "SHOT"]
```

```json
"default_ammo": "556",           // 지정된 경우 기본 탄약을 덮어씀(탄약 벨트의 경우 선택적으로 설정)
"reliability" : 8,               // 이 탄창이 얼마나 신뢰할 수 있는지 0에서 10 사이의 범위(GAME_BALANCE.md 참조)
"reload_time" : 100,             // 탄창에 탄약 한 단위를 장전하는 데 걸리는 시간
"linkage" : "ammolink"           // 설정된 경우 소비된 탄약 단위마다 하나의 연결(주어진 타입)이 떨어짐(분해 탄약 벨트용)
"reloads_like": "light_minus_battery_cell" // 이 itype를 탄창으로 받는 아이템은 이 itype도 받음
```

### 방어구

방어구는 다음과 같이 정의할 수 있습니다:

```json
"type" : "ARMOR",     // 방어구(ARMOR)로 정의합니다.
...                   // 일반 아이템과 공통되는 항목
                      // 추가적인 방어구 특정 항목:
"covers" : ["FEET"],  // 보호하는 부위. 가능한 옵션: TORSO(몸통), HEAD(머리), EYES(눈), MOUTH(입), ARMS(팔), HANDS(손), LEGS(다리), FEET(발)
"storage" : 0,        // (선택사항, 기본값 = 0) 추가되는 수납 슬롯 수
"warmth" : 10,        // (선택사항, 기본값 = 0) 옷이 제공하는 보온성
"environmental_protection" : 0,  //  (선택사항, 기본값 = 0) 환경 보호 수준(가스, 산 등에 대한 저항)
"encumbrance" : 0,    // 기본 동작 제한(수선되지 않은 상태의 값)
"max_encumbrance" : 0,    // 비강성(부드러운 소재) 수납 용기에서 용량이 가득 찼을 때의 동작 제한 값. 캐릭터 용량이 가득 차지 않은 경우 encumbrance(최소값)와 max_encumbrance(최대값) 사이의 값을 가집니다: encumbrance + (max_encumbrance - encumbrance) * character volume.
"weight_capacity_bonus": "20 kg",    // (선택사항, 기본값 = 0) 무게 용량 병행에 대한 별림치. 음수도 가능. 문자열 사용 필요 - "5000 g" 또는 "5 kg"
"weight_capacity_modifier": 1.5, // (선택사항, 기본값 = 1) 기본 무게 용량에 대한 수정 계수
"coverage" : 80,      // 신체 부위를 덮는 면적의 백분율(%)
"material_thickness" : 1,  // 재료 두께(대략 밀리미터 단위). 일반적으로 1~5, 특수 방어구는 10 이상도 가능
"power_armor" : false, // 파워 아머 여부(특수 취급됨)
"valid_mods" : ["steel_padded"] // 적용 가능한 의류 개조 목록. 의류 개조 데이터에 "제한" 항목이 없는 경우 이 목록에 작성할 필요 없음
"resistance": { "cut": 0, "bullet": 1000 }, // 설정된 경우 재료와 두께에서 계산되는 일반 방어력 계산을 덮어씁니다. 값은 무손상 상태의 아이템입니다. 두께(thickness)는 피해에 따른 성능 저하 정도에 영향을 줍니다 - 두께 1: 피해를 받아도 방어력이 감소하지 않음. 두께 2: 첫 번째 손상(1단계 피해)에서 방어력이 절반으로 감소. 두께 10: 손상 1단계마다 방어력이 10%씩 감소
"hearing_protection": 0, // 착용자가 듣는 소리를 얼마나 줄이는지 dB SPL로 지정합니다. 0~191이며 다른 소리도 포함합니다.
"adv_hearing_protection": 0 // 착용자에게 들리는 귀청이 터질 듯한 소리만 줄이는 dB SPL입니다. 다른 아이템과 누적됩니다.
```

또는 모든 아이템(책, 도구, 총, 심지어 음식도)은 armor_data를 가지면 방어구로 사용할 수 있습니다:

```json
...                   // 해당 타입에 맞는 공통 항목(예: 도구)
"armor_data" : {      // 추가로 위와 같은 방어구 데이터 항목

    "storage" : 0,
    "warmth" : 10,
```

#### 방어구 부위 데이터

여러 부위를 보호하고 부위별로 다른 커버리지나 동작 제한을 설정하려면 `armor_portion_data`를 사용합니다. 이를 통해 각 부위 또는 부위 그룹에 대해 별도의 값을 정의할 수 있습니다:

```json
"armor_portion_data": [
    {
        "covers": [ "torso" ],
        "coverage": 95,
        "encumbrance": 15
    },
    {
        "covers": [ "arms", "legs" ],
        "coverage": 80,
        "encumbrance": 10,
        "max_encumbrance": 20
    }
]
```

`armor_portion_data` 배열의 각 항목 필드:

- `covers`: 이 항목이 적용되는 신체 부위 ID 배열 (예: "torso", "head", "eyes", "mouth", "arms", "hands", "legs", "feet")
- `coverage`: 신체 부위 면적을 덮는 백분율 (0-100). 값이 높을수록 보호 성능이 높음
- `encumbrance`: 해당 부위에 대한 동작 제한 값. 기본값은 0
- `max_encumbrance`: 수납 용량이 가득 찼을 때의 동작 제한 값. 기본값은 `encumbrance`와 같음

`armor_portion_data`를 사용하는 경우, 최상위 `covers`, `coverage`, `encumbrance`, `max_encumbrance` 필드는 사용하지 않으며 부위 데이터로 대체됩니다.

### 서적

책은 다음과 같이 정의할 수 있습니다.

```json
"type" : "BOOK",      // 이것을 BOOK으로 정의합니다.
...                   // 일반 항목에 대해 위와 동일한 항목입니다.
                      // 추가 도서 관련 항목:
"max_level" : 5,      // 이 책이 훈련할 최대 기술 수준
"intelligence" : 11,  // 이 책을 페널티 없이 읽으려면 지능이 필요합니다.
"time" : "35 m",          // 단일 읽기 세션에 소요되는 시간입니다. 정수는 분 단위로 읽히거나 시간 문자열을 사용할 수 있습니다.
"fun" : -2,           // 읽기에 대한 사기 보너스/페널티
"skill" : "computer", // 스킬 상승
"chapters" : 4,       // 장 수(재미있는 책의 경우), 읽을 때마다 한 장을 "소모"합니다. 남은 장이 없는 책은 재미가 떨어집니다(내용이 이미 캐릭터에게 알려져 있기 때문입니다).
"required_level" : 2  // 학습에 필요한 최소 기술 수준
```

또는 book_data가 있는 경우 모든 항목(도구, 총, 음식까지)을 책으로 사용할 수 있습니다.

```json
"type" : "TOOL",      // 또는 다른 항목 유형
...                   // 유형과 동일한 항목(예: 모든 도구와 동일한 항목)
"book_data" : {       // 추가적으로 위와 같은 도서 데이터
    "max_level" : 5,
    "intelligence" : 11,
    "time" : "35 m",
    "fun" : -2,
    "skill" : "computer",
    "chapters" : 4,
    "use_action" : "MA_MANUAL", // book_data에는 책을 읽었을 때 트리거되는 사용 함수(USE ACTIONS 참조)가 있을 수 있습니다. 이러한 기능은 도구처럼 단순히 항목을 활성화한다고 해서 실행되는 것이 아닙니다.
    "required_level" : 2
}
```

많은 책 이름은 고유명사이므로 복수형을 명시적으로 지정해야 하는 경우가 많습니다.
다음은 책의 복수명에 대한 게임의 관례입니다.

1. 비정기간행물(교과서, 매뉴얼, 마법서 등)의 경우,
   1. 책의 단수명이 고유명사이면 복수명은 다음과 같습니다.
      `copies of (singular name)` . 예를 들어, 복수형 이름
      `Lessons for the Novice Bowhunter`은(는) `copies of Lessons for the Novice Bowhunter`입니다.
   2. 그렇지 않으면 복수형 이름은 단수형 이름의 일반적인 복수형입니다. 예를 들어, 복수형
      `tactical baton defense manual`의 이름은 `tactical baton defense manuals`입니다.
2. 정기 간행물(잡지 및 저널)의 경우,
   1. 정기간행물의 단수가 고유명사이고, "Magazine", "Weekly"로 끝나지 않는 경우,
      "월간" 등의 복수형 이름은 `issues of (singular name)` 입니다. 예를 들어, 복수형 이름
      `Archery for Kids`은(는) `issues of Archery for Kids`입니다.
   2. 그렇지 않은 경우 정기 간행물의 복수형 이름은 단수형 이름의 일반적인 복수형입니다. 예를 들어,
      `Crafty Crafter's Quarterly`의 복수형 이름은 `Crafty Crafter's Quarterlies`입니다.
3. 보드 게임의 경우(내부적으로 책 항목으로 표시됨)
   1. 보드게임의 단수명이 고유명사라면 복수형은 `sets of (singular name)` 입니다.
      예를 들어 `Picturesque`의 복수형 이름은 `sets of Picturesque`입니다.
   2. 그렇지 않으면 복수형 이름은 일반적인 복수형입니다. 예를 들어, `deck of cards`의 복수형은 다음과 같습니다.
      `decks of cards` .

#### 조건부 명명

`conditional_names` 필드를 사용하면 표시될 항목에 대한 대체 이름을 정의할 수 있습니다.
특정 조건이 충족되면 기본 이름 대신(또는 추가로). 가져 가라.
구문의 예로서 `sausage`에 대한 다음 (불완전한) 정의:

```json
{
  "name": "sausage",
  "conditional_names": [
    {
      "type": "FLAG",
      "condition": "CANNIBALISM",
      "name": "Mannwurst"
    },
    {
      "type": "COMPONENT_ID",
      "condition": "mutant",
      "name": { "str_sp": "sinister %s" }
    }
  ]
}
```

특정 항목에 대한 조건부 이름을 원하는 만큼 나열할 수 있습니다. 각 조건부 이름은 다음과 같아야 합니다.
3가지 요소로 구성됩니다.

1. 조건 유형:
   - `COMPONENT_ID`은 항목의 모든 구성 요소(및 모든 _their_ 구성 요소 등)를 검색합니다.
     on) ID에 조건 문자열이 있는 항목에 대한 것입니다. ID에는 다음 사항만 _포함_하면 됩니다.
     조건이 완벽하게 일치하지 않습니다(대소문자를 구분하지만). 예를 들어,
     조건 `mutant`은 `mutant_meat`과 일치합니다.
   - `FLAG` 항목에 지정된 플래그(정확히 일치)가 있는지 확인합니다.
2. 찾고자 하는 조건입니다.
3. 일치하는 항목이 발견된 경우 사용할 이름입니다. 유효한 표준 `name` 필드의 모든 규칙을 따릅니다.
   키는 `str` , `str_pl` 및 `ctxt` 입니다. 여기서는 %s를 사용할 수 있으며 이는 이름으로 대체됩니다.
   항목의. 이 이름 이전에 정의된 조건부 이름이 고려됩니다.

따라서 위의 예에서 소시지가 돌연변이 인간형 고기로 만들어졌다면 둘 다
`CANNIBALISM` 플래그, _and_에는 해당 ID에 `mutant`이 있는 구성 요소가 있습니다.

1. 먼저, 아이템 이름이 단수인 경우 "Mannwurst"로 완전히 대체되고, 단수인 경우에는 "Mannwursts"로 대체됩니다.
   복수.
2. 다음에는 "sinister %s"로 대체되지만 %s는 이전 이름으로 대체됩니다.
   단계로 인해 "불길한 Mannwurst"또는 "불길한 Mannwurst"가 발생합니다.

주의: `"str_sp": "sinister %s"` 대신 `"str": "sinister %s"`이 지정된 경우 복수형은
"sinister %ss"로 자동 생성되며, 이는 "sinister Mannwurstss"가 됩니다.
물론 하나의 S는 너무 멀다. 경험 법칙: 이름에 %s를 사용하는 경우 항상 동일한 이름을 지정하십시오.
당신이 무엇을 하고 있는지 정확히 알지 않는 한 복수형을 사용하세요!

#### 컬러 키

새 책을 추가할 때 다음 색상 키를 사용하세요.

- 잡지: `pink`
- “문고판” 짧은 즐거움을 주는 책(소설 포함): `light_cyan`
- “양장본” 오랫동안 즐길 수 있는 책(소설 포함): `light_blue`
- “작은 교과서” 초급 교과서, 가이드, 무술 도서: `green`
- “대형 교과서” 고급 수준의 교과서 및 고급 가이드: `blue`
- 종교 서적: `dark_gray`
- "인쇄물"(나선형 제본, 바인더 등 포함) 기술 문서(기술?)
  프로토콜, (실험실) 일지, 개인 일기: `light_green`
- 기타 독서 자료/비도서(다른 모든 카테고리가 적용되지 않는 경우에만 사용): `light_gray`

이 색상 키에 대한 몇 가지 예외가 적용될 수 있습니다(예: 겉보기와 다른 책의 경우).
BE. `yellow` 및 `red` 을 사용하지 마십시오. 해당 색상은 소리 및 적외선 비전용으로 예약되어 있습니다.

### 총

총은 다음과 같이 정의할 수 있습니다.

```json
"type": "GUN",             // 이것을 GUN으로 정의합니다
...                        // 일반 항목에 대해 위와 동일한 항목입니다.
                           // 추가 총기 관련 항목:
"skill": "pistol",         // 발사에 사용되는 스킬
"ammo": [ "357", "38" ],   // 재장전이 허용되는 탄약 유형
"ranged_damage": 0,        // 발사 시 원거리 피해
"range": 0,                // 발사 시 범위
"dispersion": 32,          // 1/4도 단위로 측정된 총의 부정확성
// gunmod에 sight_dispersion과 aim_speed가 있으면 조준 시스템은 "가장 좋은" 조준기를 선택합니다.
// 각 조준 동작에서 사용할 조준기이며, 현재 조준 한계보다 분산도가 낮은 조준기 중 가장 빠른 조준기입니다.
// 조준 한계입니다.
"sight_dispersion": 10,    // 조준 장치에서 파생된 총의 부정확성(또한 1/4도)
"aim_speed": 3,            // 분산 지점당 이동 횟수로 플레이어가 조준할 수 있는 속도를 측정한 것입니다.
"recoil": 0,               // 발사 시 반동이 발생하며 분산도는 1/4입니다.
"durability": 8,           // 손상/녹에 대한 저항성, 불발 가능성도 결정합니다.
"blackpowder_tolerance": 8,// 흑색화약 탄약 발사 시 막힐 확률(발당 1/X, 높을수록 좋음). 선택 사항이며 기본값은 8입니다.
"min_cycle_recoil": 0,     // 공격당 두 번 이상 발사할 수 있는 총의 최소 탄약 반동입니다.
"burst": 5,                // 버스트 모드에서 발사된 발사 수
"handling": 15,            // 총이 반동에서 얼마나 빨리 회복되는지에 대한 수정자입니다. 20이 기준선이고 그 이하는 더 나쁘고 그 이상은 더 좋습니다. 기본적으로 '2 - 2.0 => 0'인 스케일입니다. 예를 들어 15 핸들링은 '2 - 1.5 => 0.5' 반동 증가이고 30 핸들링은 '2 - 3.0 => -1' 반동 감소입니다.
"clip_size": 100,          // 장전할 수 있는 최대 탄약량
"ups_charges": 0,          // 일반 탄약(있는 경우) 외에도 총에는 UPS의 충전이 필요할 수 있습니다. 이는 모드에서도 작동합니다. ups_charges로 모드를 부착하면 무기의 소모량이 증가/추가됩니다.
"ammo_to_fire" 1,          // 무기에 제공될 수 있는 UPS 비용과 별도로, 발사당 사용되는 탄약의 양입니다.
// 이전 아이템 플래그 `FIRE_20`, `FIRE_50`, `FIRE_100`도 허용되며 존재하면 `ammo_to_fire`를 덮어씁니다.
"reload": 450,             // 재장전 시간, 100 = 1초 = 1 "회전". 기본값은 100입니다.
"reload_noise_volume": 6   // [사용 중단됨] 타일 거리에 따라 총을 재장전하는 소리가 얼마나 큰가요? 이는 감가상각되었으므로 대신 reload_noise_volume_dB를 사용하십시오. 이 값은 제공되는 경우 적절한 dB 볼륨으로 변환됩니다. `DEPRECIATED`
"reload_noise_volume_dB": 40, // 총을 재장전하는 소리의 크기(dB spl @1미터 기준)입니다. 기본 40dB. 일반적인 대화는 ~60dB, 귀청이 터질 듯한 소리는 120dB+
"built_in_mods": ["m203"], // IRREMOVABLE 태그를 사용하여 무기에 통합될 일련의 모드입니다.
"default_mods": ["m203"]   // 생성 시 무기에 추가될 모드의 배열입니다.
"barrel_volume": "30 mL",  // 배럴을 절단할 때 손실되는 볼륨의 양입니다. IRL 인치당 약 250ml가 적절한 근사치입니다.
"barrel_length": "30 mL",  // 명확성을 위해 대신 사용해야 하는 배럴_볼륨의 감가상각된 별칭입니다.
"valid_mod_locations": [ [ "accessories", 4 ], [ "grip", 1 ] ],  // gunmod의 유효한 위치와 해당 위치에 대한 슬롯 마운트입니다.
"loudness_modifier": 4,    // 선택적인 필드 증가 또는 감소 기본 탄약 음량(dB spl로 측정)
"speed": 100               // 초당 미터 단위로 기본 탄약 속도를 늘리거나 줄이는 선택 필드입니다. 소리의 속도는 초당 343미터로 측정됩니다.
```

```json
    "fruit_div": 2, // (선택 사항, 기본값은 1입니다). 생산된 최종 과일 요금의 양을 이 숫자로 나눕니다. 과일 아이템이 요금으로 계산되는 경우에만 작동합니다.

    "plant_name": "sunflower", // 이 씨앗에서 자라는 식물의 이름입니다. 이는 사용자에게 표시되는 정보로만 사용됩니다.
    "grow" : 91 // 기간: 식물이 완전히 성숙하는 데 걸리는 시간입니다. 더 긴 시즌 기간에 대해 더 나은 정확성을 제공하기 위해 대략 91일의 시즌 기간(거의 실제 시즌)을 기준으로 합니다.
                // 성장 시간은 나중에 Season_length 옵션에 따라 변환됩니다. 91을 기준으로 하는 것은 단지 정확성을 위한 것입니다.
                // 값 91은 3개의 전체 시즌을 의미하고, 값 30은 1개의 시즌을 의미합니다.
    "required_terrain_flag": "PLANTABLE" // 씨앗을 심을 수 있으려면 지형과 가구에 필요한 태그입니다.
```

### 연료 데이터

모든 항목 유형에는 단위당 생산되는 마력의 양을 결정하는 연료 데이터가 있을 수 있습니다.
소비되었습니다. 현재 가스와 플라즈마는 실제로 연료가 될 수 없습니다.

연료에 PERPETUAL 플래그가 있으면 해당 연료로 구동되는 엔진은 연료를 전혀 사용하지 않습니다. 이는 주로
근육 의사 연료를 위해 고안되었지만 모드는 이를 활용하여 지속적인 움직임을 만들 수 있습니다.
기계.

```json
"fuel" : {
    "energy": 34.2,               // 연료 1mL당 배터리 충전량. 배터리에는 에너지가 있습니다 1
                                 // https://en.wikipedia.org/wiki/Energy_density의 MJ/L이기도 합니다
                                 // 볼륨 1(250mL)당 스택 크기를 250으로 가정합니다. 곱하다
                                 // 250 / 스택 크기 * 다른 스택 크기의 경우 볼륨
                                 // 볼륨
   "pump_terrain": "t_gas_pump", // 선택 과목. 연료 펌프의 지형 ID입니다(있는 경우).
   "explosion_data": {           // 폭발을 일으킬 수 있는 연료의 경우 선택사항
        "chance_hot": 2,         // 1번 확률_HEAT 무기 공격 시 폭발의 뜨거움
        "chance_cold": 5,        // 1번 확률_다른 무기 공격 시 폭발의 추위
        "factor": 1.0,           // 폭발 계수 - 숫자가 클수록 더 강력한 폭발이 발생합니다.
        "fiery": true,           // 불 같은 폭발에 해당
        "size_factor": 0.1       // 크기 계수 - 숫자가 클수록 남은 연료의 폭발력이 더 커집니다.
    }
}
```

### 무작위 설명

"snippet_category" 항목이 있는 모든 항목에는 해당 조각을 기반으로 하는 임의의 설명이 있습니다.
범주:

```json
"snippet_category": "newspaper",
```

항목 설명은 다음과 같이 지정할 수 있는 조각에서 가져옵니다(
카테고리는 항목 정의의 snippet_category와 일치해야 합니다.

```json
{
  "type": "snippet",
  "category": "newspaper",
  "id": "snippet-id", // id는 선택사항이며 전문직 항목 목록에서 해당 조각이 참조될 때 사용됩니다.
  "text": "your flavor text"
}
```

또는 한 번에 여러 조각:

```json
{
  "type": "snippet",
  "category": "newspaper",
  "text": [
    "your flavor text",
    "more flavor",
    // 항목은 특정 조각을 참조하는 ID를 갖기 위해 이 형식을 사용할 수도 있습니다.
    { "id": "snippet-id", "text": "another flavor text" }
  ],
  "text": ["your flavor text", "another flavor text", "more flavor"]
}
```

동일한 카테고리에 대한 여러 스니펫이 가능하며 실제로 권장됩니다. 게임이 선택합니다
해당 유형의 각 항목에 대해 무작위로 하나.

항목 정의에 직접 조각을 넣을 수도 있습니다.

```json
"snippet_category": [ "text 1", "text 2", "text 3" ],
```

그러면 해당 항목과 관련된 스니펫 카테고리가 자동으로 생성되고 해당 카테고리가 채워집니다.
주어진 조각으로. 이 형식은 위와 같은 스니펫 ID도 지원합니다.
